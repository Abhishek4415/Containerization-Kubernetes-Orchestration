EXPERIMENT 3: DOCKER STORAGE
================================

docker rm temp-storage

docker volume create lab2-data
docker volume ls

docker run --name volume-demo -v lab2-data:/data alpine sh
docker run --rm -v lab2-data:/data alpine cat /data/record.txt

docker run --rm -v "$PWD/data:/data" alpine cat /data/record.txt

EXPERIMENT 4: DOCKER NETWORK
================================

docker network inspect lab2-network
docker inspect web-server


================================
EXPERIMENT 5: KUBERNETES DEPLOYMENT
================================

minikube start --driver=docker
minikube status

kubectl get nodes
kubectl get pods -A

minikube image load devops-flask:1.0

kubectl apply -f k8s/deployment.yaml
kubectl apply -f k8s/service.yaml

kubectl get pods
kubectl get deployments
kubectl get services


================================
EXPERIMENT 6: KUBERNETES STRATEGIES
================================

kubectl scale deployment flask-deployment --replicas=3
kubectl scale deployment flask-deployment --replicas=5
kubectl scale deployment flask-deployment --replicas=2

docker build -t devops-flask:2.0 .
minikube image load devops-flask:2.0

kubectl set image deployment/flask-deployment flask-app=devops-flask:2.0
kubectl rollout status deployment/flask-deployment
kubectl rollout history deployment/flask-deployment
kubectl rollout undo deployment/flask-deployment
